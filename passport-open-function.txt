===== index.html =====

async function abrirPasaporte() {
  if (!_pasConfirmado) {
    toast('Selecciona un paciente de la lista primero', 'warn');
    return;
  }

  let link = (_pasCurrent && _pasCurrent.link) || document.getElementById('pasLinkTexto').textContent;
  let popup = null;

  if (!link) {
    const btn = document.getElementById('pasAbrirBtn');
    popup = window.open('', '_blank');
    if (btn) {
      btn.disabled = true;
      btn.textContent = 'Generando enlace...';
      btn.style.cursor = 'wait';
    }
    try {
      await generarLinkPasaporte();
    } catch (error) {
      toast('No se pudo generar el enlace seguro', 'error');
    } finally {
      if (btn) {
        btn.disabled = false;
        btn.textContent = 'Abrir y editar →';
        btn.style.cursor = 'pointer';
      }
    }
    link = (_pasCurrent && _pasCurrent.link) || document.getElementById('pasLinkTexto').textContent;
  }

  if (!link) {
    if (popup && !popup.closed) popup.close();
    return;
  }

  if (popup && !popup.closed) popup.location.replace(link);
  else window.open(link, '_blank');
}
